\section{Khả năng tái lập thực nghiệm, Phần cứng và Chi phí thực tế}
\label{app:repro}

\noindent \textbf{Lệnh tái lập một bước (\texttt{D4 / R8}).} Toàn bộ mã nguồn kiểm thử ngoại tuyến, kiểm định cấu hình và tái tạo bảng kết quả được chạy từ thư mục gốc repository (khóa phụ thuộc tại \texttt{pyproject.toml}, mẫu biến môi trường tại \texttt{.env.example}):

\begin{lstlisting}[language=bash, caption={Lệnh cài đặt, kiểm chứng và tái lập thực nghiệm một bước}]
# 1. Cai dat moi truong va chay toan bo kiem thu offline (khong can API key)
python -m pip install -e ".[dev,web,mcp]"
ruff check .
pytest -q

# 2. Tham dinh cau hinh, chay self-test replay va tai tao bang tong hop
python -m evaluation.runner validate --experiment evaluation/experiments/d3-gemini.yaml
python -m evaluation.runner run --experiment evaluation/experiments/d3-gemini.yaml \
  --adapter replay --output artifacts/d3/replay-selftest
python -m evaluation.runner summarize --input artifacts/d3/<run-id>

# 3. One-command reproduction script (D4)
python scripts/reproduce_d3.py
\end{lstlisting}

\section{Tuyên bố sử dụng AI (AI-Use Statement)}
\label{app:ai-use}

\noindent Nhóm sử dụng công cụ AI ở hai vai trò tách biệt và hoàn toàn minh bạch:
\begin{enumerate}
    \item \textbf{Công cụ hỗ trợ phát triển và soạn thảo:} sử dụng trợ lý lập trình (như Claude Code / Antigravity) để gợi ý cấu trúc hàm kiểm thử, tái cấu trúc kịch bản đánh giá và hỗ trợ rà soát văn phong LaTeX. Mọi mã nguồn, quy tắc phân quyền và nội dung kỹ thuật đều do thành viên phụ trách trực tiếp đọc hiểu, đối chiếu mã nguồn và chạy kiểm thử xác nhận trước khi commit.
    \item \textbf{Thành phần trong hệ thống được đánh giá:} sử dụng \texttt{gemini-2.5-flash} và \texttt{qwen3} (Ollama cục bộ) làm bộ sinh SQL (\textit{Planner}) và tác nhân đối chứng \texttt{NaiveAgent}. Không sử dụng bất kỳ LLM nào để sinh, ước lượng hay sửa đổi số liệu thực nghiệm.
\end{enumerate}
